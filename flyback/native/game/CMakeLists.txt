set(FBAR_GAME_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
get_filename_component(FBAR_REPO "${FBAR_GAME_DIR}/../../.." ABSOLUTE)
set(FBAR_GAME_DATA "${CMAKE_BINARY_DIR}/data" CACHE PATH "Staged map files and prebuilt official players")
set(FBAR_OFFICIAL_BOTS
    gpt-6.2-astra gpt-6.2-sol claude-fable-5.2 claude-opus-5.5 gemini-4-carbon
    gemini-4-hydrogen grok-4.7 glm-5.3 deepseek-v4.1-flash kimi-k3.1-pro qwen4-max mimo-v2.6-pro doubao-seed-evolving)
set(FBAR_LEGACY_MAPS twinlakes yinyang icefloes ruins skybridge stonecircle)
set(FBAR_FORK_MAPS trident beacon tidelock kiln windrose moonwell)
foreach(map IN LISTS FBAR_LEGACY_MAPS FBAR_FORK_MAPS)
    if(map IN_LIST FBAR_FORK_MAPS)
        set(src "${FBAR_GAME_DIR}/maps/${map}.txt")
    else()
        set(src "${FBAR_REPO}/boomerang/renderer/maps/${map}.txt")
    endif()
    configure_file("${src}" "${FBAR_GAME_DATA}/maps/${map}.txt" COPYONLY)
    install(FILES "${src}" DESTINATION data/maps)
endforeach()
add_library(flyback_game STATIC game.cpp bot.cpp)
target_compile_features(flyback_game PUBLIC cxx_std_17)
target_include_directories(flyback_game PUBLIC "${FBAR_GAME_DIR}" "${FBAR_GAME_DIR}/..")
target_link_libraries(flyback_game PUBLIC nlohmann_json::nlohmann_json)
if(WIN32)
    target_sources(flyback_game PRIVATE win_process.cpp)
    target_link_libraries(flyback_game PRIVATE userenv advapi32 psapi)
    target_compile_definitions(flyback_game PRIVATE _WIN32_WINNT=0x0A00 NOMINMAX WIN32_LEAN_AND_MEAN)
    set(FBAR_PLAYER_SUFFIX .exe)
    set(FBAR_PLAYER_LINK -Wl,--stack,67108864)
else()
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|amd64|AMD64")
        message(FATAL_ERROR "Dedicated referee currently supports Windows x64 and Linux x86_64")
    endif()
    target_sources(flyback_game PRIVATE linux_process.cpp)
    find_package(Threads REQUIRED)
    target_link_libraries(flyback_game PRIVATE Threads::Threads)
    add_executable(fbar_bot_guard linux_guard.cpp)
    set_target_properties(fbar_bot_guard PROPERTIES OUTPUT_NAME arena-bot-guard RUNTIME_OUTPUT_DIRECTORY "${FBAR_GAME_DATA}/bots")
    install(TARGETS fbar_bot_guard RUNTIME DESTINATION data/bots)
    install(PROGRAMS run-delegated DESTINATION .)
    install(FILES flyback.service DESTINATION examples)
    install(FILES README-Linux.md DESTINATION .)
endif()
if(MSVC)
    target_compile_options(flyback_game PUBLIC /utf-8)
endif()
option(FBAR_BUILD_OFFICIAL_BOTS "Compile and stage all thirteen competition entries" ON)
if(FBAR_BUILD_OFFICIAL_BOTS)
    if(MSVC)
        find_program(FBAR_PLAYER_CXX NAMES g++ REQUIRED)
        if(NOT FBAR_PLAYER_CXX)
            message(FATAL_ERROR "Official Windows players require MinGW-w64 g++, including MiMo's __int128")
        endif()
    else()
        set(FBAR_PLAYER_CXX "${CMAKE_CXX_COMPILER}")
    endif()
    file(MAKE_DIRECTORY "${FBAR_GAME_DATA}/bots")
    execute_process(COMMAND "${FBAR_PLAYER_CXX}" -dumpmachine OUTPUT_VARIABLE FBAR_BOT_TRIPLE OUTPUT_STRIP_TRAILING_WHITESPACE)
    execute_process(COMMAND "${FBAR_PLAYER_CXX}" -dumpversion OUTPUT_VARIABLE FBAR_BOT_COMPILER_VERSION OUTPUT_STRIP_TRAILING_WHITESPACE)
    configure_file(build-info.json.in "${FBAR_GAME_DATA}/bots/build-info.json" @ONLY)
    install(FILES "${FBAR_GAME_DATA}/bots/build-info.json" DESTINATION data/bots)
    foreach(bot IN LISTS FBAR_OFFICIAL_BOTS)
        set(src "${FBAR_REPO}/boomerang/bots/${bot}/player.cpp")
        set(out "${FBAR_GAME_DATA}/bots/${bot}${FBAR_PLAYER_SUFFIX}")
        add_custom_command(OUTPUT "${out}" COMMAND "${FBAR_PLAYER_CXX}" -std=c++17 -O2 -static ${FBAR_PLAYER_LINK} "-I${FBAR_GAME_DIR}" "${src}" -o "${out}" DEPENDS "${src}" "${FBAR_GAME_DIR}/dart.hpp" VERBATIM)
        list(APPEND FBAR_BOT_OUTPUTS "${out}")
    endforeach()
    add_custom_target(flyback_bots ALL DEPENDS ${FBAR_BOT_OUTPUTS})
    install(PROGRAMS ${FBAR_BOT_OUTPUTS} DESTINATION data/bots)
endif()
if(BUILD_TESTING)
    add_executable(flyback_game_test tests/game_test.cpp)
    target_link_libraries(flyback_game_test PRIVATE flyback_game)
    add_test(NAME flyback_game COMMAND flyback_game_test "${FBAR_GAME_DATA}")
endif()
