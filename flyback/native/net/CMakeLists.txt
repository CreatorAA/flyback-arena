if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(flyback_transport LANGUAGES CXX)
  include(CTest)
  include(FetchContent)
  if(NOT TARGET nlohmann_json::nlohmann_json)
    FetchContent_Declare(net_json URL https://codeload.github.com/nlohmann/json/tar.gz/refs/tags/v3.12.0)
    FetchContent_MakeAvailable(net_json)
  endif()
endif()
include(FetchContent)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
find_package(zstd CONFIG REQUIRED)
find_package(Threads REQUIRED)
set(FLYBACK_MSQUIC_SDK "" CACHE PATH "Windows: unpacked native NuGet build/native directory; Linux: optional SDK prefix")
set(FLYBACK_MDNS_INCLUDE "" CACHE PATH "Directory with pinned mjansson/mdns.h")
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Flyback requires x64 Windows")
  endif()
  if(NOT FLYBACK_MSQUIC_SDK)
    FetchContent_Declare(fbar_msquic URL https://api.nuget.org/v3-flatcontainer/microsoft.native.quic.msquic.openssl/2.6.4/microsoft.native.quic.msquic.openssl.2.6.4.nupkg DOWNLOAD_NAME msquic.zip)
    FetchContent_MakeAvailable(fbar_msquic)
    set(FLYBACK_MSQUIC_SDK "${fbar_msquic_SOURCE_DIR}/build/native")
  endif()
  add_library(fbar_quic_runtime SHARED IMPORTED GLOBAL)
  set_target_properties(fbar_quic_runtime PROPERTIES IMPORTED_LOCATION "${FLYBACK_MSQUIC_SDK}/bin/x64/msquic.dll" IMPORTED_IMPLIB "${FLYBACK_MSQUIC_SDK}/lib/x64/msquic.lib" INTERFACE_INCLUDE_DIRECTORIES "${FLYBACK_MSQUIC_SDK}/include")
  install(FILES "${FLYBACK_MSQUIC_SDK}/bin/x64/msquic.dll" DESTINATION .)
  install(FILES "${FLYBACK_MSQUIC_SDK}/../../LICENSE" "${FLYBACK_MSQUIC_SDK}/../../THIRD-PARTY-NOTICES" DESTINATION licenses/msquic OPTIONAL)
else()
  find_library(FBAR_MSQUIC_LIBRARY NAMES msquic HINTS "${FLYBACK_MSQUIC_SDK}/lib")
  if(NOT FBAR_MSQUIC_LIBRARY)
    # Runtime distributions have only the SONAME, not the libmsquic.so linker symlink.
    find_file(FBAR_MSQUIC_LIBRARY NAMES libmsquic.so.2 HINTS "${FLYBACK_MSQUIC_SDK}/lib" PATHS "/usr/lib/${CMAKE_LIBRARY_ARCHITECTURE}" "/lib/${CMAKE_LIBRARY_ARCHITECTURE}" /usr/lib64 /usr/lib REQUIRED)
  endif()
  find_path(FBAR_MSQUIC_INCLUDE NAMES msquic.h HINTS "${FLYBACK_MSQUIC_SDK}/include")
  if(NOT FBAR_MSQUIC_INCLUDE)
    # The upstream runtime-only .deb has no headers.
    set(FBAR_MSQUIC_INCLUDE "${CMAKE_CURRENT_BINARY_DIR}/msquic-include")
    file(MAKE_DIRECTORY "${FBAR_MSQUIC_INCLUDE}")
    foreach(f msquic.h msquic_posix.h quic_sal_stub.h)
      if(NOT EXISTS "${FBAR_MSQUIC_INCLUDE}/${f}")
        file(DOWNLOAD "https://raw.githubusercontent.com/microsoft/msquic/v2.6.4/src/inc/${f}" "${FBAR_MSQUIC_INCLUDE}/${f}.part" STATUS download_status TLS_VERIFY ON)
        list(GET download_status 0 error)
        if(error)
          message(FATAL_ERROR "MsQuic header ${f}: ${download_status}")
        endif()
        file(RENAME "${FBAR_MSQUIC_INCLUDE}/${f}.part" "${FBAR_MSQUIC_INCLUDE}/${f}")
      endif()
    endforeach()
  endif()
  add_library(fbar_quic_runtime SHARED IMPORTED GLOBAL)
  set_target_properties(fbar_quic_runtime PROPERTIES IMPORTED_LOCATION "${FBAR_MSQUIC_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${FBAR_MSQUIC_INCLUDE}")
endif()
if(NOT FLYBACK_MDNS_INCLUDE)
  FetchContent_Declare(fbar_mdns_source URL https://codeload.github.com/mjansson/mdns/tar.gz/a569c4759bd47e0f2a7bfc4d4c19620445782806 SOURCE_SUBDIR fbar_header_only)
  FetchContent_MakeAvailable(fbar_mdns_source)
  set(FLYBACK_MDNS_INCLUDE "${fbar_mdns_source_SOURCE_DIR}")
  install(FILES "${fbar_mdns_source_SOURCE_DIR}/LICENSE" DESTINATION licenses/mdns OPTIONAL)
endif()
add_library(flyback_net STATIC net.cpp identity.cpp wire.cpp discovery.cpp)
target_compile_features(flyback_net PUBLIC cxx_std_17)
target_include_directories(flyback_net PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${FLYBACK_MDNS_INCLUDE}")
target_link_libraries(flyback_net PUBLIC nlohmann_json::nlohmann_json fbar_quic_runtime PRIVATE OpenSSL::Crypto zstd::libzstd_shared Threads::Threads)
if(WIN32)
  target_link_libraries(flyback_net PRIVATE ws2_32 ntdll advapi32 iphlpapi)
endif()

  if(WIN32)

  endif()
