cmake_minimum_required(VERSION 3.23)
project(FlybackArena VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(MSVC)
  add_compile_options(/utf-8 /EHsc /bigobj)
endif()
if(WIN32)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
endif()
include(CTest)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
set(FLYBACK_HEADERS "${CMAKE_BINARY_DIR}/_deps/headers" CACHE PATH "Pinned header cache; can be populated offline")
function(fbar_header name repo version source)
  if(NOT EXISTS "${FLYBACK_HEADERS}/${name}")
    get_filename_component(dir "${FLYBACK_HEADERS}/${name}" DIRECTORY)
    file(MAKE_DIRECTORY "${dir}")
    foreach(base "https://raw.githubusercontent.com/${repo}/${version}/${source}" "https://cdn.jsdelivr.net/gh/${repo}@${version}/${source}")
      file(DOWNLOAD "${base}" "${FLYBACK_HEADERS}/${name}.part" TLS_VERIFY ON STATUS result TIMEOUT 90 INACTIVITY_TIMEOUT 20)
      list(GET result 0 rc)
      if(rc EQUAL 0)
        file(RENAME "${FLYBACK_HEADERS}/${name}.part" "${FLYBACK_HEADERS}/${name}")
        break()
      endif()
    endforeach()
    if(NOT EXISTS "${FLYBACK_HEADERS}/${name}")
      message(FATAL_ERROR "Could not obtain ${name}: ${result}. Populate FLYBACK_HEADERS with pinned source.")
    endif()
  endif()
endfunction()
fbar_header(nlohmann/json.hpp nlohmann/json v3.12.0 single_include/nlohmann/json.hpp)
fbar_header(httplib.h yhirose/cpp-httplib v0.20.1 httplib.h)
fbar_header(toml.hpp marzer/tomlplusplus v3.4.0 toml.hpp)
add_library(fbar_json INTERFACE)
target_include_directories(fbar_json SYSTEM INTERFACE "${FLYBACK_HEADERS}")
add_library(nlohmann_json::nlohmann_json ALIAS fbar_json)
add_subdirectory(native/game)
add_subdirectory(native/net)
add_library(flyback_host native/common/util.cpp native/host/runner.cpp native/host/room.cpp native/host/service.cpp native/admin/web.cpp)
target_include_directories(flyback_host PUBLIC native)
target_link_libraries(flyback_host PUBLIC flyback_game flyback_net nlohmann_json::nlohmann_json OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
target_compile_definitions(flyback_host PRIVATE CPPHTTPLIB_OPENSSL_SUPPORT)
if(WIN32)
  target_link_libraries(flyback_host PUBLIC ws2_32 crypt32)
else()
  find_library(ARGON2_LIB NAMES argon2 libargon2.so.1)
  if(ARGON2_LIB)
    target_compile_definitions(flyback_host PRIVATE FLYBACK_ARGON2_LIB)
    target_link_libraries(flyback_host PRIVATE "${ARGON2_LIB}")
  endif()
endif()
add_executable(arena-server native/host/server_main.cpp)
if(MINGW)
  target_link_options(arena-server PRIVATE -municode)
endif()
add_executable(flyback-bridge native/host/bridge.cpp native/host/replay.cpp)
target_link_libraries(arena-server PRIVATE flyback_host)
target_link_libraries(flyback-bridge PRIVATE flyback_host)
if(WIN32)
  target_link_libraries(flyback_host PRIVATE psapi)
  foreach(app arena-server flyback-bridge)
    add_custom_command(TARGET ${app} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:fbar_quic_runtime>" "$<TARGET_FILE_DIR:${app}>")
  endforeach()
endif()
install(TARGETS arena-server flyback-bridge RUNTIME DESTINATION .)
install(DIRECTORY admin-web/ DESTINATION admin-web OPTIONAL)
install(FILES packaging/server.toml DESTINATION .)
